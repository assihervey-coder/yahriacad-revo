     1→     1→/**
     2→     2→ * NEXUS PCB — Routeur maze A* multicouche
     3→     3→ * Équivalent : services/router/topological/ + geometrical/ + via_placer/
     4→     4→ *
     5→     5→ * Stratégie :
     6→     6→ *   1. Ordre de routage par criticité (RF > différentiel > haute vitesse > alim > signal)
     7→     7→ *   2. Routage en arbre multi-broches : Dijkstra/A* multi-sources depuis l'arbre
     8→     8→ *      existant du net vers chaque broche restante
     9→     9→ *   3. Obstacles : corps de composants (couche top), pistes étrangères dilatées
    10→    10→ *      selon leur largeur (clearance), marge de bord, keepout RF
    11→    11→ *   4. Changement de couche = via (coût 14) — le via_placer les pose au point
    12→    12→ *      exact de transition
    13→    13→ *   5. Nets infranchissables → signalés au DRC (boucle de rétroaction)
    14→    14→ */

    16→    16→  Constraint, DesignRules, Netlist, PlacedComponent, Route, RoutingSolution, TraceSegment, Via,

    18→    18→import { padWorldPos, placedRect } from './world-model'

    20→    20→const RES = 0.25 // résolution de grille (mm)
    21→    21→
    22→    22→class MinHeap {
    23→    23→  private keys: number[] = []
    24→    24→  private nodes: number[] = []
    25→    25→  get size() { return this.nodes.length }
    26→    26→  push(node: number, key: number) {
    27→    27→    this.keys.push(key); this.nodes.push(node)
    28→    28→    let i = this.nodes.length - 1
    29→    29→    while (i > 0) {
    30→    30→      const p = (i - 1) >> 1
    31→    31→      if (this.keys[p] <= this.keys[i]) break
    32→    32→      this.swap(i, p); i = p
    33→    33→    }
    34→    34→  }
    35→    35→  pop(): number {
    36→    36→    const top = this.nodes[0]
    37→    37→    const lastK = this.keys.pop()!, lastN = this.nodes.pop()!
    38→    38→    if (this.nodes.length > 0) {
    39→    39→      this.keys[0] = lastK; this.nodes[0] = lastN
    40→    40→      let i = 0
    41→    41→      for (;;) {
    42→    42→        const l = 2 * i + 1, r = l + 1
    43→    43→        let m = i
    44→    44→        if (l < this.keys.length && this.keys[l] < this.keys[m]) m = l
    45→    45→        if (r < this.keys.length && this.keys[r] < this.keys[m]) m = r
    46→    46→        if (m === i) break
    47→    47→        this.swap(i, m); i = m
    48→    48→      }
    49→    49→    }
    50→    50→    return top
    51→    51→  }
    52→    52→  private swap(a: number, b: number) {
    53→    53→    ;[this.keys[a], this.keys[b]] = [this.keys[b], this.keys[a]]
    54→    54→    ;[this.nodes[a], this.nodes[b]] = [this.nodes[b], this.nodes[a]]
    55→    55→  }
    56→    56→}

    63→    63→const CLASS_ORDER: Record<string, number> = {
    64→    64→  rf: 2, highspeed: 3, diffpair: 4, power: 1, ground: 0, analog: 5, signal: 6,

    66→    66→
    67→    67→export function routeAll(
    68→    68→  nl: Netlist, placements: PlacedComponent[], rules: DesignRules,
    69→    69→  constraints: Constraint[], opts: RouterOptions = {},
    70→    70→): RoutingSolution {
    71→    71→  const t0 = Date.now()
    72→    72→  const cols = Math.ceil(nl.board.w / RES)
    73→    73→  const rows = Math.ceil(nl.board.h / RES)
    74→    74→  const L = 2 // couches cuivre
    75→    75→  const cells = cols * rows
    76→    76→  const total = cells * L
    77→    77→
    78→    78→  const idxOf = (x: number, y: number) => y * cols + x
    79→    79→
    80→    80→  /* ---- Masques d'obstacles ---- */
    81→    81→  const edgeCells = Math.ceil(rules.edgeClearance / RES)
    82→    82→  const bodyBlocked = new Uint8Array(total)     // corps de composants (couche top)
    83→    83→  const padNet = new Int16Array(total).fill(-1) // cellule-pad (couche 0) → index de net propriétaire
    84→    84→  const padNetL1 = new Int16Array(total).fill(-1) // réserve d'échappatoire L1 au-dessus des pads
    85→    85→  const keepBlock = new Uint8Array(total)       // keepout RF (2 couches)
    86→    86→  const occupied = new Int16Array(total).fill(-1) // cellules cuivre prises
    87→    87→  const dilated = new Int16Array(total).fill(-1)  // zone de clearance interdite
    88→    88→
    89→    89→  const netIndex = new Map<string, number>()
    90→    90→  nl.nets.forEach((n, i) => netIndex.set(n.name, i))
    91→    91→  const compByRef = new Map(nl.components.map((c) => [c.ref, c]))
    92→    92→
    93→    93→  // Marge de bord
    94→    94→  for (let y = 0; y < rows; y++)
    95→    95→    for (let x = 0; x < cols; x++)
    96→    96→      if (x < edgeCells || y < edgeCells || x >= cols - edgeCells || y >= rows - edgeCells) {
    97→    97→        bodyBlocked[idxOf(x, y)] = 1
    98→    98→        bodyBlocked[cells + idxOf(x, y)] = 1
    99→    99→      }
   100→   100→
   101→   101→  // Corps de composants (couche 0 uniquement : on peut router dessous en couche 1)
   102→   102→  for (const p of placements) {
   103→   103→    const c = compByRef.get(p.ref)!
   104→   104→    const r = placedRect(p, c)
   105→   105→    const x0 = Math.max(0, Math.floor((r.x - r.w / 2) / RES)), x1 = Math.min(cols - 1, Math.ceil((r.x + r.w / 2) / RES))
   106→   106→    const y0 = Math.max(0, Math.floor((r.y - r.h / 2) / RES)), y1 = Math.min(rows - 1, Math.ceil((r.y + r.h / 2) / RES))
   107→   107→    for (let y = y0; y <= y1; y++)
   108→   108→      for (let x = x0; x <= x1; x++)
   109→   109→        bodyBlocked[idxOf(x, y)] = 1
   110→   110→  }
   111→   111→
   112→   112→  // Pads : cellules autorisées pour leur net propriétaire
   113→   113→  const padCell = new Map<string, number>() // "ref.pin" → idx
   114→   114→  for (const p of placements) {
   115→   115→    const c = compByRef.get(p.ref)!
   116→   116→    for (const pd of c.footprint.pads) {
   117→   117→      const w = padWorldPos(p, pd)
   118→   118→      const x = Math.min(cols - 1, Math.max(0, Math.round(w.x / RES)))
   119→   119→      const y = Math.min(rows - 1, Math.max(0, Math.round(w.y / RES)))
   120→   120→      const i = idxOf(x, y)
   121→   121→      const netName = c.pins[pd.pin]
   122→   122→      if (!netName) continue
   123→   123→      const ni = netIndex.get(netName)
   124→   124→      if (ni === undefined) continue
   125→   125→      bodyBlocked[i] = 0            // un pad n'est jamais un obstacle pour son net
   126→   126→      if (padNet[i] === -1) padNet[i] = ni
   127→   127→      // Réserve d'échappatoire : le pad reste dégagé sur L1 tant que son net
   128→   128→      // n'est pas routé (libéré dynamiquement une fois le net connecté)
   129→   129→      if (padNetL1[i] === -1) padNetL1[i] = ni
   130→   130→      padCell.set(`${p.ref}.${pd.pin}`, i)
   131→   131→    }
   132→   132→  }
   133→   133→
   134→   134→  // Keepout RF (autour des composants marqués keepout)
   135→   135→  for (const cst of constraints) {
   136→   136→    if (cst.kind !== 'keepout') continue
   137→   137→    for (const ref of cst.refs) {
   138→   138→      const c = compByRef.get(ref); const p = placements.find((x) => x.ref === ref)
   139→   139→      if (!c || !p) continue
   140→   140→      const r = placedRect(p, c)
   141→   141→      const m = (cst.value ?? 5)
   142→   142→      const x0 = Math.max(0, Math.floor((r.x - r.w / 2 - m) / RES)), x1 = Math.min(cols - 1, Math.ceil((r.x + r.w / 2 + m) / RES))
   143→   143→      const y0 = Math.max(0, Math.floor((r.y - r.h / 2 - m) / RES)), y1 = Math.min(rows - 1, Math.ceil((r.y + r.h / 2 + m) / RES))
   144→   144→      for (let y = y0; y <= y1; y++)
   145→   145→        for (let x = x0; x <= x1; x++) {
   146→   146→          // keepout RF sur la couche supérieure uniquement : le plan de masse
   147→   147→          // en couche inférieure sous l'antenne est une bonne pratique RF
   148→   148→          keepBlock[idxOf(x, y)] = 1
   149→   149→        }
   150→   150→    }
   151→   151→  }
   152→   152→
   153→   153→  const isRfNet = (ni: number) => nl.nets[ni]?.cls === 'rf'
   154→   154→  const widthOfNet = (ni: number) => {
   155→   155→    const cls = nl.nets[ni].cls
   156→   156→    return nl.nets[ni].width ?? rules.widths[cls] ?? 0.25
   157→   157→  }
   158→   158→
   159→   159→  /* ---- Marquage d'une piste routée (occupation + dilation) ---- */
   160→   160→  function markPath(path: number[], ni: number, vias: { x: number; y: number }[]) {
   161→   161→    const w = widthOfNet(ni)
   162→   162→    // rayon de dilation : mon bord + demi-largeur du net voisin le plus fin + clearance
   163→   163→    const radius = Math.max(1, Math.ceil((w / 2 + rules.minTraceWidth / 2 + rules.clearance) / RES) - 1)
   164→   164→    const viaRadius = Math.max(radius, Math.ceil((0.9 / 2 + rules.minTraceWidth / 2 + rules.clearance) / RES) - 1)
   165→   165→    const mark = (i: number, radius: number) => {
   166→   166→      occupied[i] = ni
   167→   167→      const x = i % cols, y = Math.floor(i / cols) % rows
   168→   168→      const layer = i < cells ? 0 : 1
   169→   169→      for (let dy = -radius; dy <= radius; dy++)
   170→   170→        for (let dx = -radius; dx <= radius; dx++) {
   171→   171→          const nx = x + dx, ny = y + dy
   172→   172→          if (nx < 0 || ny < 0 || nx >= cols || ny >= rows) continue
   173→   173→          const j = layer * cells + idxOf(nx, ny)
   174→   174→          if (dilated[j] === -1) dilated[j] = ni
   175→   175→        }
   176→   176→    }
   177→   177→    for (const i of path) mark(i, radius)
   178→   178→    for (const v of vias) {
   179→   179→      const x = Math.min(cols - 1, Math.max(0, Math.round(v.x / RES)))
   180→   180→      const y = Math.min(rows - 1, Math.max(0, Math.round(v.y / RES)))
   181→   181→      mark(idxOf(x, y), viaRadius); mark(cells + idxOf(x, y), viaRadius)
   182→   182→    }
   183→   183→  }
   184→   184→
   185→   185→  /* ---- A* multi-sources : arbre existant → broche cible ---- */
   186→   186→  const gStamp = new Int32Array(total)
   187→   187→  const gVal = new Float32Array(total)
   188→   188→  const parent = new Int32Array(total)
   189→   189→  let stamp = 0
   190→   190→
   191→   191→  const routedSet = new Set<number>()
   192→   192→
   193→   193→  /** Coût d'une transition de couche (via) — majoré par le via_minimizer [DeepPCB] */
   194→   194→  let viaCost = 14
   195→   195→
   196→   196→  function passable(i: number, ni: number, relaxed = false): boolean {
   197→   197→    if (occupied[i] !== -1 && occupied[i] !== ni) return false
   198→   198→    if (!relaxed && dilated[i] !== -1 && dilated[i] !== ni) return false
   199→   199→    if (padNet[i] !== -1 && padNet[i] !== ni) return false
   200→   200→    // Échappatoire L1 au-dessus des pads : réservée aux nets non encore routés
   201→   201→    if (i >= cells) {
   202→   202→      const p1 = padNetL1[i - cells]
   203→   203→      if (p1 !== -1 && p1 !== ni && !routedSet.has(p1)) return false
   204→   204→    }
   205→   205→    if (keepBlock[i] && !isRfNet(ni) && padNet[i] !== ni) return false
   206→   206→    return true
   207→   207→  }

   209→   209→  function search(tree: Set<number>, source: number, ni: number, relaxed = false): number[] | null {
   210→   210→    stamp++
   211→   211→    const heap = new MinHeap()
   212→   212→    const sb = source % cells
   213→   213→    const sx = sb % cols, sy = Math.floor(sb / cols)
   214→   214→    const h = (i: number) => {
   215→   215→      const b = i % cells
   216→   216→      const x = b % cols, y = Math.floor(b / cols)
   217→   217→      return Math.abs(x - sx) + Math.abs(y - sy)
   218→   218→    }
   219→   219→    for (const t of tree) {
   220→   220→      gStamp[t] = stamp; gVal[t] = 0; parent[t] = -1
   221→   221→      heap.push(t, h(t))
   222→   222→    }
   223→   223→    const VIA_COST = viaCost
   224→   224→    let expansions = 0
   225→   225→    while (heap.size > 0) {
   226→   226→      const cur = heap.pop()
   227→   227→      if (gStamp[cur] !== stamp) continue
   228→   228→      if (cur === source) {
   229→   229→        // reconstruction
   230→   230→        const path: number[] = []
   231→   231→        let n = cur
   232→   232→        while (n !== -1) { path.push(n); n = parent[n] }
   233→   233→        return path.reverse()
   234→   234→      }
   235→   235→      if (++expansions > 150000) return null
   236→   236→      const layer = cur < cells ? 0 : 1
   237→   237→      const base = cur % cells
   238→   238→      const x = base % cols, y = Math.floor(base / cols)
   239→   239→      const layerBase = layer * cells
   240→   240→      const neighbors: [number, number][] = []
   241→   241→      // Mouvements latéraux SUR LA MÊME COUCHE (coût uniforme)
   242→   242→      if (x > 0) neighbors.push([layerBase + base - 1, 1])
   243→   243→      if (x < cols - 1) neighbors.push([layerBase + base + 1, 1])
   244→   244→      if (y > 0) neighbors.push([layerBase + base - cols, 1])
   245→   245→      if (y < rows - 1) neighbors.push([layerBase + base + cols, 1])
   246→   246→      // Via : couche opposée, même cellule
   247→   247→      neighbors.push([cells - layerBase + base, VIA_COST])
   248→   248→      for (const [nb, cost] of neighbors) {
   249→   249→        if (nb < 0 || nb >= total) continue
   250→   250→        const layerNb = nb < cells ? 0 : 1
   251→   251→        const isVia = layerNb !== layer
   252→   252→        if (isVia) {
   253→   253→          // un via ne se pose jamais dans le corps d'un composant (couche top)
   254→   254→          if (bodyBlocked[nb] && padNet[nb] !== ni) continue
   255→   255→        } else if (layerNb === 0 && bodyBlocked[nb] && padNet[nb] !== ni) {
   256→   256→          continue // piste sur le cuivre supérieur : corps interdit
   257→   257→        }
   258→   258→        if (!passable(nb, ni, relaxed)) continue
   259→   259→        const ng = gVal[cur] + cost
   260→   260→        if (gStamp[nb] !== stamp || ng < gVal[nb]) {
   261→   261→          gStamp[nb] = stamp; gVal[nb] = ng; parent[nb] = cur
   262→   262→          heap.push(nb, ng + h(nb))
   263→   263→        }
   264→   264→      }
   265→   265→    }
   266→   266→    if (process.env.NEXUS_DEBUG) {
   267→   267→      const b = source % cells
   268→   268→      console.log(`[search-fail] net=${ni} src=(${b % cols},${Math.floor(b / cols)}) expansions=${expansions} treeSize=${tree.size}`)
   269→   269→      // diagnostic : pourquoi le pad de l'arbre est-il scellé ?
   270→   270→      const seedCell = [...tree][0]
   271→   271→      const sb2 = seedCell % cells
   272→   272→      const sx2 = sb2 % cols, sy2 = Math.floor(sb2 / cols)
   273→   273→      const reason = (i: number) => {
   274→   274→        const parts: string[] = []
   275→   275→        if (occupied[i] !== -1 && occupied[i] !== ni) parts.push(`occ=${occupied[i]}`)
   276→   276→        if (dilated[i] !== -1 && dilated[i] !== ni) parts.push(`dil=${dilated[i]}`)
   277→   277→        if (padNet[i] !== -1 && padNet[i] !== ni) parts.push(`pad=${padNet[i]}`)
   278→   278→        if (bodyBlocked[i] && padNet[i] !== ni) parts.push('body')
   279→   279→        if (keepBlock[i] && !isRfNet(ni)) parts.push('keep')
   280→   280→        return parts.join(',') || 'OK'
   281→   281→      }
   282→   282→      console.log(`  seed=(${sx2},${sy2}) layer0 raison=${reason(seedCell)}`)
   283→   283→      for (const [dx, dy] of [[-1, 0], [1, 0], [0, -1], [0, 1]] as const) {
   284→   284→        const nx = sx2 + dx, ny = sy2 + dy
   285→   285→        if (nx < 0 || ny < 0 || nx >= cols || ny >= rows) continue
   286→   286→        console.log(`  L0 voisin (${nx},${ny}): ${reason(idxOf(nx, ny))} | L1: ${reason(cells + idxOf(nx, ny))}`)
   287→   287→      }
   288→   288→    }
   289→   289→    return null
   290→   290→  }

   292→   292→  /* ---- Boucle principale : routage glouton + RIP-UP & REROUTE ---- */
   293→   293→  // Cellules-pad pré-calculées par net (avant le tri : spanOf en dépend)
   294→   294→  const netPinsCellCache = new Map<string, number[]>()
   295→   295→  for (const net of nl.nets) {
   296→   296→    const pinCells: number[] = []
   297→   297→    for (const { ref, pin } of net.pins) {
   298→   298→      const i = padCell.get(`${ref}.${pin}`)
   299→   299→      if (i !== undefined && i >= 0) pinCells.push(i)
   300→   300→    }
   301→   301→    netPinsCellCache.set(net.name, [...new Set(pinCells)])
   302→   302→  }
   303→   303→  const netPinsCells = new Map<number, number[]>()
   304→   304→  for (const net of nl.nets) {
   305→   305→    netPinsCells.set(netIndex.get(net.name)!, netPinsCellCache.get(net.name)!)
   306→   306→  }
   307→   307→
   308→   308→  // Ordre : criticité de classe, puis empan croissant (les bus courts d'abord)
   309→   309→  const spanOf = (n: (typeof nl.nets)[number]) => {
   310→   310→    const cells2 = netPinsCellCache.get(n.name)
   311→   311→    if (!cells2 || cells2.length === 0) return 0
   312→   312→    let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity
   313→   313→    for (const c of cells2) {
   314→   314→      const x = c % cols, y = Math.floor(c / cols)
   315→   315→      if (x < minX) minX = x; if (x > maxX) maxX = x
   316→   316→      if (y < minY) minY = y; if (y > maxY) maxY = y
   317→   317→    }
   318→   318→    return maxX - minX + (maxY - minY)
   319→   319→  }
   320→   320→  const ordered = [...nl.nets]
   321→   321→    .filter((n) => n.pins.length >= 2)
   322→   322→    .sort((a, b) => (CLASS_ORDER[a.cls] ?? 9) - (CLASS_ORDER[b.cls] ?? 9) || spanOf(a) - spanOf(b))

   328→   328→    lengthMm: number
   329→   329→  }
   330→   330→  const routedStore = new Map<number, NetRouteState>()
   331→   331→  const failedNets = new Set<number>()
   332→   332→  const triedRip = new Map<number, Set<number>>()
   333→   333→  let done = 0

   335→   335→  /** Reconstruit les masques d'occupation depuis l'état des nets routés */
   336→   336→  function rebuildMasks() {
   337→   337→    occupied.fill(-1)
   338→   338→    dilated.fill(-1)
   339→   339→    routedSet.clear()
   340→   340→    for (const k of routedStore.keys()) routedSet.add(k)
   341→   341→    for (const [ni2, st] of routedStore) markPath(st.tree, ni2, st.vias)

   344→   344→  /** Tente de router entièrement un net (croissance d'arbre, broches les plus proches d'abord) */
   345→   345→  function attemptRoute(ni: number): { ok: boolean; blockedAt?: number } {
   346→   346→    const net = nl.nets[ni]
   347→   347→    const pinCells = netPinsCells.get(ni)!
   348→   348→    const w = widthOfNet(ni)
   349→   349→    const tree = new Set<number>([pinCells[0], pinCells[0] + cells])
   350→   350→    const allSegments: TraceSegment[] = []
   351→   351→    const allVias: Via[] = []
   352→   352→    let netLen = 0
   353→   353→    const remaining = pinCells.slice(1)
   354→   354→
   355→   355→    const nearestFirst = () => {
   356→   356→      let best = 0, bestD = Infinity
   357→   357→      const tcs = [...tree].filter((t) => t < cells)
   358→   358→      for (let i = 0; i < remaining.length; i++) {
   359→   359→        const bx = remaining[i] % cols, by = Math.floor(remaining[i] / cols)
   360→   360→        let d = Infinity
   361→   361→        for (const t of tcs) {
   362→   362→          const ddx = Math.abs((t % cols) - bx), ddy = Math.abs(Math.floor(t / cols) - by)
   363→   363→          if (ddx + ddy < d) d = ddx + ddy
   364→   364→        }
   365→   365→        if (d < bestD) { bestD = d; best = i }
   366→   366→      }
   367→   367→      return best
   368→   368→    }
   369→   369→
   370→   370→    const bx = (v: number) => ({ x: (v % cols) * RES + RES / 2, y: Math.floor((v % cells) / cols) * RES + RES / 2 })
   371→   371→
   372→   372→    while (remaining.length > 0) {
   373→   373→      const pick = nearestFirst()
   374→   374→      const target = remaining.splice(pick, 1)[0]
   375→   375→      // masse & puissance : tentative relâchée d'emblée (elles ont le droit de passer
   376→   376→      // dans les zones de clearance des autres, jamais sur leur cuivre)
   377→   377→      const relaxedFirst = net.cls === 'ground' || net.cls === 'power'
   378→   378→      let path = relaxedFirst ? search(tree, target, ni, true) : search(tree, target, ni)
   379→   379→      if (!path) path = search(tree, target, ni, true) // repli : clearance relâchée
   380→   380→      if (!path) {
   381→   381→        if (process.env.NEXUS_DEBUG) {
   382→   382→          const net2 = nl.nets[ni]
   383→   383→          const pin = net2.pins.find(({ ref, pin: p }) => padCell.get(`${ref}.${p}`) === target)
   384→   384→          const bx2 = target % cols, by2 = Math.floor(target / cols)
   385→   385→          const why = (i: number, relaxed2: boolean) => {
   386→   386→            const parts: string[] = []
   387→   387→            if (occupied[i] !== -1 && occupied[i] !== ni) parts.push(`occ=${occupied[i]}`)
   388→   388→            if (!relaxed2 && dilated[i] !== -1 && dilated[i] !== ni) parts.push(`dil=${dilated[i]}`)
   389→   389→            if (padNet[i] !== -1 && padNet[i] !== ni) parts.push(`pad=${padNet[i]}`)
   390→   390→            if (i >= cells) {
   391→   391→              const p1 = padNetL1[i - cells]
   392→   392→              if (p1 !== -1 && p1 !== ni && !routedSet.has(p1)) parts.push(`res=${p1}`)
   393→   393→            }
   394→   394→            if (keepBlock[i] && !isRfNet(ni) && padNet[i] !== ni) parts.push('keep')
   395→   395→            return parts.join(',') || 'OK'
   396→   396→          }
   397→   397→          console.log(`[route-fail] net=${net2.name} broche=${pin ? `${pin.ref}.${pin.pin}` : '?'} cellule=(${bx2},${by2})`)
   398→   398→          for (const [dx, dy] of [[-1, 0], [1, 0], [0, -1], [0, 1]] as const) {
   399→   399→            const nx2 = bx2 + dx, ny2 = by2 + dy
   400→   400→            if (nx2 < 0 || ny2 < 0 || nx2 >= cols || ny2 >= rows) continue
   401→   401→            console.log(`   L0(${nx2},${ny2}):${why(idxOf(nx2, ny2), true)} | L1:${why(cells + idxOf(nx2, ny2), true)}`)
   402→   402→          }
   403→   403→        }
   404→   404→        return { ok: false, blockedAt: target }
   405→   405→      }
   406→   406→      for (const i of path) tree.add(i)
   407→   407→      // Construction des segments + vias
   408→   408→      let curLayer: 0 | 1 = path[0] < cells ? 0 : 1
   409→   409→      let pts: { x: number; y: number }[] = []
   410→   410→      const flush = (endLayer: 0 | 1) => {
   411→   411→        if (pts.length >= 2) {
   412→   412→          const simplified: { x: number; y: number }[] = [pts[0]]
   413→   413→          for (let i = 1; i < pts.length - 1; i++) {
   414→   414→            const a = simplified[simplified.length - 1], b = pts[i], c = pts[i + 1]
   415→   415→            const collinear = (a.x === b.x && b.x === c.x) || (a.y === b.y && b.y === c.y)
   416→   416→            if (!collinear) simplified.push(b)
   417→   417→          }
   418→   418→          simplified.push(pts[pts.length - 1])
   419→   419→          let len = 0
   420→   420→          for (let i = 1; i < simplified.length; i++)
   421→   421→            len += Math.abs(simplified[i].x - simplified[i - 1].x) + Math.abs(simplified[i].y - simplified[i - 1].y)
   422→   422→          netLen += len
   423→   423→          allSegments.push({ net: net.name, layer: endLayer, pts: simplified, width: w })
   424→   424→        }
   425→   425→      }
   426→   426→      for (let i = 1; i < path.length; i++) {
   427→   427→        const prev = path[i - 1], curr = path[i]
   428→   428→        const layerPrev: 0 | 1 = prev < cells ? 0 : 1
   429→   429→        const layerCur: 0 | 1 = curr < cells ? 0 : 1
   430→   430→        if (layerCur !== layerPrev) {
   431→   431→          pts.push(bx(prev))
   432→   432→          flush(layerPrev)
   433→   433→          const pos = bx(prev)
   434→   434→          allVias.push({ net: net.name, x: Math.round(pos.x * 100) / 100, y: Math.round(pos.y * 100) / 100, drill: rules.minDrill, diameter: rules.minDrill + 0.5 })
   435→   435→          curLayer = layerCur
   436→   436→          pts = [bx(curr)]
   437→   437→        } else {
   438→   438→          if (pts.length === 0) pts.push(bx(prev))
   439→   439→          pts.push(bx(curr))
   440→   440→        }
   441→   441→      }
   442→   442→      flush(curLayer)
   443→   443→    }
   444→   444→
   445→   445→    const seenVia = new Set<string>()
   446→   446→    const uniqVias = allVias.filter((v) => {
   447→   447→      const k = `${v.x},${v.y}`
   448→   448→      if (seenVia.has(k)) return false
   449→   449→      seenVia.add(k); return true
   450→   450→    })
   451→   451→    markPath([...tree], ni, uniqVias)
   452→   452→    routedSet.add(ni)
   453→   453→    routedStore.set(ni, { segments: allSegments, vias: uniqVias, tree: [...tree], lengthMm: netLen })
   454→   454→    return { ok: true }
   455→   455→  }
   456→   456→
   457→   457→  /** Trouve le net routé dont le cuivre bloque le plus la broche en échec
   458→   458→   *  (en excluant les rip-ups déjà tentés et échoués pour ce net) */
   459→   459→  function findBlocker(fi: number, blockedAt: number): number {
   460→   460→    const bx2 = blockedAt % cols, by2 = Math.floor(blockedAt / cols)
   461→   461→    const tried = triedRip.get(fi) ?? new Set<number>()
   462→   462→    let best = -1, bestD = Infinity
   463→   463→    for (const [ri, st] of routedStore) {
   464→   464→      if (ri === fi || tried.has(ri)) continue
   465→   465→      for (let i = 0; i < st.tree.length; i += 2) {
   466→   466→        const c = st.tree[i] % cells
   467→   467→        const d = Math.abs((c % cols) - bx2) + Math.abs(Math.floor(c / cols) - by2)
   468→   468→        if (d < bestD) { bestD = d; best = ri }
   469→   469→      }
   470→   470→    }
   471→   471→    return best
   472→   472→  }
   473→   473→
   474→   474→  /* ---- Passe 1 : routage glouton par criticité ---- */
   475→   475→  for (const net of ordered) {
   476→   476→    if (opts.shouldCancel?.()) break
   477→   477→    const ni = netIndex.get(net.name)!
   478→   478→    const pinCells = netPinsCells.get(ni)!
   479→   479→    if (pinCells.length < 2) {
   480→   480→      done++
   481→   481→      opts.onProgress?.({ done, total: ordered.length, net: net.name, ok: true })
   482→   482→      continue
   483→   483→    }
   484→   484→    const r = attemptRoute(ni)
   485→   485→    if (!r.ok) failedNets.add(ni)
   486→   486→    done++
   487→   487→    opts.onProgress?.({ done, total: ordered.length, net: net.name, ok: r.ok })
   488→   488→  }
   489→   489→
   490→   490→  /* ---- Passe 2 : RIP-UP & REROUTE (jusqu'à 8 rounds, rip multi-bloquants) ---- */
   491→   491→  for (let round = 0; round < 8 && failedNets.size > 0; round++) {
   492→   492→    if (opts.shouldCancel?.()) break
   493→   493→    let progress = false
   494→   494→    for (const fi of [...failedNets]) {
   495→   495→      if (opts.shouldCancel?.()) break
   496→   496→      const probe = attemptRoute(fi)
   497→   497→      const blockedAt = probe.blockedAt
   498→   498→      if (blockedAt === undefined) { failedNets.delete(fi); triedRip.delete(fi); progress = true; continue }
   499→   499→      const blocker = findBlocker(fi, blockedAt)
   500→   500→      if (blocker === -1) continue
   501→   501→      const snapshot = new Map(routedStore)
   502→   502→      const blockerNet = nl.nets[blocker].name
   503→   503→      routedStore.delete(blocker)
   504→   504→      rebuildMasks()
   505→   505→      const rF = attemptRoute(fi)
   506→   506→      if (rF.ok) {
   507→   507→        failedNets.delete(fi)
   508→   508→        triedRip.delete(fi)
   509→   509→        opts.onProgress?.({ done, total: ordered.length, net: `${nl.nets[fi].name} (rip-up de ${blockerNet})`, ok: true })
   510→   510→        const rB = attemptRoute(blocker)
   511→   511→        if (!rB.ok) failedNets.add(blocker)
   512→   512→        progress = true
   513→   513→      } else {
   514→   514→        // Échec : mémorise ce bloquant pour tenter le suivant au prochain round
   515→   515→        if (!triedRip.has(fi)) triedRip.set(fi, new Set())
   516→   516→        triedRip.get(fi)!.add(blocker)
   517→   517→        routedStore.clear()
   518→   518→        for (const [k, v] of snapshot) routedStore.set(k, v)
   519→   519→        rebuildMasks()
   520→   520→      }
   521→   521→    }
   522→   522→    if (!progress) break
   523→   523→  }
   524→   524→
   525→   525→  /* ---- Passe 2b : MINIMISATION DES VIAS [DeepPCB via_minimizer] ----
   526→   526→   * Chaque net routé possédant des vias est re-routé avec un coût de via
   527→   527→   * fortement majoré : si une variante avec moins de transitions de couche
   528→   528→   * existe dans l'espace libre actuel, elle remplace la première passe.
   529→   529→   * Garde-fou : longueur acceptée jusqu'à +30 % (+2 mm) — un via coûte plus
   530→   530→   * cher qu'un détour (fiabilité, insertion, fabrication). */
   531→   531→  let viasRemoved = 0
   532→   532→  {
   533→   533→    viaCost = 46
   534→   534→    const candidates = [...routedStore.keys()].filter((ni) => routedStore.get(ni)!.vias.length > 0)
   535→   535→    for (const ni of candidates) {
   536→   536→      if (opts.shouldCancel?.()) break
   537→   537→      const before = routedStore.get(ni)!
   538→   538→      routedStore.delete(ni)
   539→   539→      rebuildMasks()
   540→   540→      const r = attemptRoute(ni)
   541→   541→      const after = routedStore.get(ni)
   542→   542→      if (r.ok && after && after.vias.length < before.vias.length && after.lengthMm <= before.lengthMm * 1.3 + 2) {
   543→   543→        viasRemoved += before.vias.length - after.vias.length
   544→   544→        opts.onProgress?.({ done, total: ordered.length, net: `${nl.nets[ni].name} (−${before.vias.length - after.vias.length} via)`, ok: true })
   545→   545→      } else {
   546→   546→        // Variante pas assez bonne → restauration de la route d'origine
   547→   547→        routedStore.set(ni, before)
   548→   548→        rebuildMasks()
   549→   549→      }
   550→   550→    }
   551→   551→    viaCost = 14
   552→   552→  }
   553→   553→
   554→   554→  /* ---- Passe 3 : plan de masse synthétique (copper pour B.Cu) ----
   555→   555→   * Les nets de masse irroutables sont connectés par un plan cuivre généré
   556→   556→   * sur la couche inférieure : toutes les cellules libres, inondées depuis
   557→   557→   * les pads de masse, forment un plan continu (pratique industrielle 2 couches). */
   558→   558→  let groundPour: RoutingSolution['groundPour'] | undefined
   559→   559→  for (const fi of [...failedNets]) {
   560→   560→    if (nl.nets[fi].cls !== 'ground') continue
   561→   561→    const pins = netPinsCells.get(fi)!
   562→   562→    const niG = fi
   563→   563→    // Inondation bicouche : cellules libres L0+L1 reliées par transitions de couche
   564→   564→    const freeCell = (node: number): boolean => {
   565→   565→      const layer = node < cells ? 0 : 1
   566→   566→      const base = node % cells
   567→   567→      if (layer === 0) {
   568→   568→        if (occupied[base] !== -1 || bodyBlocked[base]) return false
   569→   569→        const pn = padNet[base]
   570→   570→        return pn === -1 || pn === niG
   571→   571→      }
   572→   572→      return occupied[cells + base] === -1
   573→   573→    }
   574→   574→    const start0 = pins[0]
   575→   575→    if (!freeCell(start0) || !freeCell(cells + start0)) continue
   576→   576→    const region = new Set<number>([start0, cells + start0])
   577→   577→    const queue: number[] = [start0, cells + start0]
   578→   578→    while (queue.length > 0) {
   579→   579→      const cur = queue.pop()!
   580→   580→      const layer = cur < cells ? 0 : 1
   581→   581→      const base = cur % cells
   582→   582→      const cx = base % cols, cy = Math.floor(base / cols)
   583→   583→      const neighbors: number[] = []
   584→   584→      if (cx > 0) neighbors.push(cur - 1)
   585→   585→      if (cx < cols - 1) neighbors.push(cur + 1)
   586→   586→      if (cy > 0) neighbors.push(cur - cols)
   587→   587→      if (cy < rows - 1) neighbors.push(cur + cols)
   588→   588→      neighbors.push(layer === 0 ? cells + base : base) // via
   589→   589→      for (const nb of neighbors) {
   590→   590→        if (nb < 0 || nb >= total) continue
   591→   591→        if (region.has(nb) || !freeCell(nb)) continue
   592→   592→        region.add(nb)
   593→   593→        queue.push(nb)
   594→   594→      }
   595→   595→    }
   596→   596→    // Tous les pads de masse doivent appartenir au plan
   597→   597→    if (!pins.every((p) => region.has(p))) {
   598→   598→      if (process.env.NEXUS_DEBUG) {
   599→   599→        const reached = pins.filter((p) => region.has(p)).length
   600→   600→        console.log(`[pour] net=${nl.nets[fi].name} région bicouche=${region.size} cellules, pads joints=${reached}/${pins.length} — ÎLOTS SÉPARÉS`)
   601→   601→      }
   602→   602→      continue
   603→   603→    }
   604→   604→    failedNets.delete(fi)
   605→   605→    const toMm = (node: number) => {
   606→   606→      const base = node % cells
   607→   607→      return {
   608→   608→        x: Math.round(((base % cols) * RES + RES / 2) * 100) / 100,
   609→   609→        y: Math.round((Math.floor(base / cols) * RES + RES / 2) * 100) / 100,
   610→   610→      }
   611→   611→    }
   612→   612→    groundPour = {
   613→   613→      top: [...region].filter((n) => n < cells).map(toMm),
   614→   614→      bottom: [...region].filter((n) => n >= cells).map(toMm),
   615→   615→      cols, rows, res: RES,
   616→   616→    }
   617→   617→    opts.onProgress?.({ done, total: ordered.length, net: `${nl.nets[fi].name} (plan de masse synthétique bicouche)`, ok: true })
   618→   618→  }
   619→   619→
   620→   620→  /* ---- Résultats finaux ---- */
   621→   621→  const routes: Route[] = []
   622→   622→  let routedCount = 0
   623→   623→  let totalLen = 0
   624→   624→  let viaCount = 0
   625→   625→  for (const net of ordered) {
   626→   626→    const ni = netIndex.get(net.name)!
   627→   627→    const pinCells = netPinsCells.get(ni)!
   628→   628→    const st = routedStore.get(ni)
   629→   629→    if (st) {
   630→   630→      routedCount++
   631→   631→      totalLen += st.lengthMm
   632→   632→      viaCount += st.vias.length
   633→   633→      routes.push({ net: net.name, segments: st.segments, vias: st.vias, lengthMm: Math.round(st.lengthMm * 10) / 10, routed: true })
   634→   634→    } else if (groundPour && nl.nets[ni].cls === 'ground') {
   635→   635→      routedCount++
   636→   636→      routes.push({ net: net.name, segments: [], vias: [], lengthMm: 0, routed: true, pour: true })
   637→   637→    } else if (pinCells.length < 2) {
   638→   638→      routedCount++
   639→   639→      routes.push({ net: net.name, segments: [], vias: [], lengthMm: 0, routed: true })
   640→   640→    } else {
   641→   641→      routes.push({ net: net.name, segments: [], vias: [], lengthMm: 0, routed: false, failureReason: 'Aucun chemin libre — rip-up épuisé' })
   642→   642→    }
   643→   643→  }
   644→   644→
   645→   645→  return {
   646→   646→    routes,
   647→   647→    routedNets: routedCount,
   648→   648→    totalNets: ordered.length,
   649→   649→    totalLengthMm: Math.round(totalLen * 10) / 10,
   650→   650→    viaCount,
   651→   651→    viasRemoved,
   652→   652→    groundPour,
   653→   653→    durationMs: Date.now() - t0,
   654→   654→  }
   655→   655→}
