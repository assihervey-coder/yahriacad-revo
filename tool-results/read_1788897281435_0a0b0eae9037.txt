     1→     1→/**
     2→     2→ * NEXUS PCB — Routeur maze A* multicouche
     3→     3→ * Équivalent : services/router/topological/ + geometrical/ + via_placer/
     4→     4→ *
     5→     5→ * Stratégie :
     6→     6→ *   1. Ordre de routage par criticité (RF > différentiel > haute vitesse > alim > signal)
     7→     7→ *   2. Routage en arbre multi-broches : Dijkstra/A* multi-sources depuis l'arbre
     8→     8→ *      existant du net vers chaque broche restante
     9→     9→ *   3. Obstacles : corps de composants (couche top), pistes étrangères dilatées
    10→    10→ *      selon leur largeur (clearance), marge de bord, keepout RF
    11→    11→ *   4. Changement de couche = via (coût 14) — le via_placer les pose au point
    12→    12→ *      exact de transition
    13→    13→ *   5. Nets infranchissables → signalés au DRC (boucle de rétroaction)
    14→    14→ */

    16→    16→  Constraint, DesignRules, Netlist, PlacedComponent, Route, RoutingSolution, TraceSegment, Via,

    18→    18→import { padWorldPos, placedRect } from './world-model'

    20→    20→const RES = 0.25 // résolution de grille (mm)
    21→    21→
    22→    22→class MinHeap {
    23→    23→  private keys: number[] = []
    24→    24→  private nodes: number[] = []
    25→    25→  get size() { return this.nodes.length }
    26→    26→  push(node: number, key: number) {
    27→    27→    this.keys.push(key); this.nodes.push(node)
    28→    28→    let i = this.nodes.length - 1
    29→    29→    while (i > 0) {
    30→    30→      const p = (i - 1) >> 1
    31→    31→      if (this.keys[p] <= this.keys[i]) break
    32→    32→      this.swap(i, p); i = p
    33→    33→    }
    34→    34→  }
    35→    35→  pop(): number {
    36→    36→    const top = this.nodes[0]
    37→    37→    const lastK = this.keys.pop()!, lastN = this.nodes.pop()!
    38→    38→    if (this.nodes.length > 0) {
    39→    39→      this.keys[0] = lastK; this.nodes[0] = lastN
    40→    40→      let i = 0
    41→    41→      for (;;) {
    42→    42→        const l = 2 * i + 1, r = l + 1
    43→    43→        let m = i
    44→    44→        if (l < this.keys.length && this.keys[l] < this.keys[m]) m = l
    45→    45→        if (r < this.keys.length && this.keys[r] < this.keys[m]) m = r
    46→    46→        if (m === i) break
    47→    47→        this.swap(i, m); i = m
    48→    48→      }
    49→    49→    }
    50→    50→    return top
    51→    51→  }
    52→    52→  private swap(a: number, b: number) {
    53→    53→    ;[this.keys[a], this.keys[b]] = [this.keys[b], this.keys[a]]
    54→    54→    ;[this.nodes[a], this.nodes[b]] = [this.nodes[b], this.nodes[a]]
    55→    55→  }
    56→    56→}

    63→    63→const CLASS_ORDER: Record<string, number> = {
    64→    64→  rf: 2, highspeed: 3, diffpair: 4, power: 1, ground: 0, analog: 5, signal: 6,

    66→    66→
    67→    67→export function routeAll(
    68→    68→  nl: Netlist, placements: PlacedComponent[], rules: DesignRules,
    69→    69→  constraints: Constraint[], opts: RouterOptions = {},
    70→    70→): RoutingSolution {
    71→    71→  const t0 = Date.now()
    72→    72→  const cols = Math.ceil(nl.board.w / RES)
    73→    73→  const rows = Math.ceil(nl.board.h / RES)
    74→    74→  const L = 2 // couches cuivre
    75→    75→  const cells = cols * rows
    76→    76→  const total = cells * L
    77→    77→
    78→    78→  const idxOf = (x: number, y: number) => y * cols + x
    79→    79→
    80→    80→  /* ---- Masques d'obstacles ---- */
    81→    81→  const edgeCells = Math.ceil(rules.edgeClearance / RES)
    82→    82→  const bodyBlocked = new Uint8Array(total)     // corps de composants (couche top)
    83→    83→  const padNet = new Int16Array(total).fill(-1) // cellule-pad (couche 0) → index de net propriétaire
    84→    84→  const padNetL1 = new Int16Array(total).fill(-1) // réserve d'échappatoire L1 au-dessus des pads
    85→    85→  const keepBlock = new Uint8Array(total)       // keepout RF (2 couches)
    86→    86→  const occupied = new Int16Array(total).fill(-1) // cellules cuivre prises
    87→    87→  const dilated = new Int16Array(total).fill(-1)  // zone de clearance interdite
    88→    88→
    89→    89→  const netIndex = new Map<string, number>()
    90→    90→  nl.nets.forEach((n, i) => netIndex.set(n.name, i))
    91→    91→  const compByRef = new Map(nl.components.map((c) => [c.ref, c]))
    92→    92→
    93→    93→  // Marge de bord
    94→    94→  for (let y = 0; y < rows; y++)
    95→    95→    for (let x = 0; x < cols; x++)
    96→    96→      if (x < edgeCells || y < edgeCells || x >= cols - edgeCells || y >= rows - edgeCells) {
    97→    97→        bodyBlocked[idxOf(x, y)] = 1
    98→    98→        bodyBlocked[cells + idxOf(x, y)] = 1
    99→    99→      }
   100→   100→
   101→   101→  // Corps de composants (couche 0 uniquement : on peut router dessous en couche 1)
   102→   102→  for (const p of placements) {
   103→   103→    const c = compByRef.get(p.ref)!
   104→   104→    const r = placedRect(p, c)
   105→   105→    const x0 = Math.max(0, Math.floor((r.x - r.w / 2) / RES)), x1 = Math.min(cols - 1, Math.ceil((r.x + r.w / 2) / RES))
   106→   106→    const y0 = Math.max(0, Math.floor((r.y - r.h / 2) / RES)), y1 = Math.min(rows - 1, Math.ceil((r.y + r.h / 2) / RES))
   107→   107→    for (let y = y0; y <= y1; y++)
   108→   108→      for (let x = x0; x <= x1; x++)
   109→   109→        bodyBlocked[idxOf(x, y)] = 1
   110→   110→  }
   111→   111→
   112→   112→  // Pads : cellules autorisées pour leur net propriétaire
   113→   113→  const padCell = new Map<string, number>() // "ref.pin" → idx
   114→   114→  for (const p of placements) {
   115→   115→    const c = compByRef.get(p.ref)!
   116→   116→    for (const pd of c.footprint.pads) {
   117→   117→      const w = padWorldPos(p, pd)
   118→   118→      const x = Math.min(cols - 1, Math.max(0, Math.round(w.x / RES)))
   119→   119→      const y = Math.min(rows - 1, Math.max(0, Math.round(w.y / RES)))
   120→   120→      const i = idxOf(x, y)
   121→   121→      const netName = c.pins[pd.pin]
   122→   122→      if (!netName) continue
   123→   123→      const ni = netIndex.get(netName)
   124→   124→      if (ni === undefined) continue
   125→   125→      bodyBlocked[i] = 0            // un pad n'est jamais un obstacle pour son net
   126→   126→      if (padNet[i] === -1) padNet[i] = ni
   127→   127→      // Réserve d'échappatoire : le pad reste dégagé sur L1 tant que son net
   128→   128→      // n'est pas routé (libéré dynamiquement une fois le net connecté)
   129→   129→      if (padNetL1[i] === -1) padNetL1[i] = ni
   130→   130→      padCell.set(`${p.ref}.${pd.pin}`, i)
   131→   131→    }
   132→   132→  }
   133→   133→
   134→   134→  // Keepout RF (autour des composants marqués keepout)
   135→   135→  for (const cst of constraints) {
   136→   136→    if (cst.kind !== 'keepout') continue
   137→   137→    for (const ref of cst.refs) {
   138→   138→      const c = compByRef.get(ref); const p = placements.find((x) => x.ref === ref)
   139→   139→      if (!c || !p) continue
   140→   140→      const r = placedRect(p, c)
   141→   141→      const m = (cst.value ?? 5)
   142→   142→      const x0 = Math.max(0, Math.floor((r.x - r.w / 2 - m) / RES)), x1 = Math.min(cols - 1, Math.ceil((r.x + r.w / 2 + m) / RES))
   143→   143→      const y0 = Math.max(0, Math.floor((r.y - r.h / 2 - m) / RES)), y1 = Math.min(rows - 1, Math.ceil((r.y + r.h / 2 + m) / RES))
   144→   144→      for (let y = y0; y <= y1; y++)
   145→   145→        for (let x = x0; x <= x1; x++) {
   146→   146→          // keepout RF sur la couche supérieure uniquement : le plan de masse
   147→   147→          // en couche inférieure sous l'antenne est une bonne pratique RF
   148→   148→          keepBlock[idxOf(x, y)] = 1
   149→   149→        }
   150→   150→    }
   151→   151→  }
   152→   152→
   153→   153→  const isRfNet = (ni: number) => nl.nets[ni]?.cls === 'rf'
   154→   154→  const widthOfNet = (ni: number) => {
   155→   155→    const cls = nl.nets[ni].cls
   156→   156→    return nl.nets[ni].width ?? rules.widths[cls] ?? 0.25
   157→   157→  }
   158→   158→
   159→   159→  /* ---- Marquage d'une piste routée (occupation + dilation) ---- */
   160→   160→  function markPath(path: number[], ni: number, vias: { x: number; y: number }[]) {
   161→   161→    const w = widthOfNet(ni)
   162→   162→    // rayon de dilation : mon bord + demi-largeur du net voisin le plus fin + clearance
   163→   163→    const radius = Math.max(1, Math.ceil((w / 2 + rules.minTraceWidth / 2 + rules.clearance) / RES) - 1)
   164→   164→    const viaRadius = Math.max(radius, Math.ceil((0.9 / 2 + rules.minTraceWidth / 2 + rules.clearance) / RES) - 1)
   165→   165→    const mark = (i: number, radius: number) => {
   166→   166→      occupied[i] = ni
   167→   167→      const x = i % cols, y = Math.floor(i / cols) % rows
   168→   168→      const layer = i < cells ? 0 : 1
   169→   169→      for (let dy = -radius; dy <= radius; dy++)
   170→   170→        for (let dx = -radius; dx <= radius; dx++) {
   171→   171→          const nx = x + dx, ny = y + dy
   172→   172→          if (nx < 0 || ny < 0 || nx >= cols || ny >= rows) continue
   173→   173→          const j = layer * cells + idxOf(nx, ny)
   174→   174→          if (dilated[j] === -1) dilated[j] = ni
   175→   175→        }
   176→   176→    }
   177→   177→    for (const i of path) mark(i, radius)
   178→   178→    for (const v of vias) {
   179→   179→      const x = Math.min(cols - 1, Math.max(0, Math.round(v.x / RES)))
   180→   180→      const y = Math.min(rows - 1, Math.max(0, Math.round(v.y / RES)))
   181→   181→      mark(idxOf(x, y), viaRadius); mark(cells + idxOf(x, y), viaRadius)
   182→   182→    }
   183→   183→  }
   184→   184→
   185→   185→  /* ---- A* multi-sources : arbre existant → broche cible ---- */
   186→   186→  const gStamp = new Int32Array(total)
   187→   187→  const gVal = new Float32Array(total)
   188→   188→  const parent = new Int32Array(total)
   189→   189→  let stamp = 0
   190→   190→
   191→   191→  const routedSet = new Set<number>()
   192→   192→
   193→   193→  function passable(i: number, ni: number, relaxed = false): boolean {
   194→   194→    if (occupied[i] !== -1 && occupied[i] !== ni) return false
   195→   195→    if (!relaxed && dilated[i] !== -1 && dilated[i] !== ni) return false
   196→   196→    if (padNet[i] !== -1 && padNet[i] !== ni) return false
   197→   197→    // Échappatoire L1 au-dessus des pads : réservée aux nets non encore routés
   198→   198→    if (i >= cells) {
   199→   199→      const p1 = padNetL1[i - cells]
   200→   200→      if (p1 !== -1 && p1 !== ni && !routedSet.has(p1)) return false
   201→   201→    }
   202→   202→    if (keepBlock[i] && !isRfNet(ni) && padNet[i] !== ni) return false
   203→   203→    return true
   204→   204→  }

   206→   206→  function search(tree: Set<number>, source: number, ni: number, relaxed = false): number[] | null {
   207→   207→    stamp++
   208→   208→    const heap = new MinHeap()
   209→   209→    const sb = source % cells
   210→   210→    const sx = sb % cols, sy = Math.floor(sb / cols)
   211→   211→    const h = (i: number) => {
   212→   212→      const b = i % cells
   213→   213→      const x = b % cols, y = Math.floor(b / cols)
   214→   214→      return Math.abs(x - sx) + Math.abs(y - sy)
   215→   215→    }
   216→   216→    for (const t of tree) {
   217→   217→      gStamp[t] = stamp; gVal[t] = 0; parent[t] = -1
   218→   218→      heap.push(t, h(t))
   219→   219→    }
   220→   220→    const VIA_COST = 14
   221→   221→    let expansions = 0
   222→   222→    while (heap.size > 0) {
   223→   223→      const cur = heap.pop()
   224→   224→      if (gStamp[cur] !== stamp) continue
   225→   225→      if (cur === source) {
   226→   226→        // reconstruction
   227→   227→        const path: number[] = []
   228→   228→        let n = cur
   229→   229→        while (n !== -1) { path.push(n); n = parent[n] }
   230→   230→        return path.reverse()
   231→   231→      }
   232→   232→      if (++expansions > 150000) return null
   233→   233→      const layer = cur < cells ? 0 : 1
   234→   234→      const base = cur % cells
   235→   235→      const x = base % cols, y = Math.floor(base / cols)
   236→   236→      const layerBase = layer * cells
   237→   237→      const neighbors: [number, number][] = []
   238→   238→      // Mouvements latéraux SUR LA MÊME COUCHE (coût uniforme)
   239→   239→      if (x > 0) neighbors.push([layerBase + base - 1, 1])
   240→   240→      if (x < cols - 1) neighbors.push([layerBase + base + 1, 1])
   241→   241→      if (y > 0) neighbors.push([layerBase + base - cols, 1])
   242→   242→      if (y < rows - 1) neighbors.push([layerBase + base + cols, 1])
   243→   243→      // Via : couche opposée, même cellule
   244→   244→      neighbors.push([cells - layerBase + base, VIA_COST])
   245→   245→      for (const [nb, cost] of neighbors) {
   246→   246→        if (nb < 0 || nb >= total) continue
   247→   247→        const layerNb = nb < cells ? 0 : 1
   248→   248→        const isVia = layerNb !== layer
   249→   249→        if (isVia) {
   250→   250→          // un via ne se pose jamais dans le corps d'un composant (couche top)
   251→   251→          if (bodyBlocked[nb] && padNet[nb] !== ni) continue
   252→   252→        } else if (layerNb === 0 && bodyBlocked[nb] && padNet[nb] !== ni) {
   253→   253→          continue // piste sur le cuivre supérieur : corps interdit
   254→   254→        }
   255→   255→        if (!passable(nb, ni, relaxed)) continue
   256→   256→        const ng = gVal[cur] + cost
   257→   257→        if (gStamp[nb] !== stamp || ng < gVal[nb]) {
   258→   258→          gStamp[nb] = stamp; gVal[nb] = ng; parent[nb] = cur
   259→   259→          heap.push(nb, ng + h(nb))
   260→   260→        }
   261→   261→      }
   262→   262→    }
   263→   263→    if (process.env.NEXUS_DEBUG) {
   264→   264→      const b = source % cells
   265→   265→      console.log(`[search-fail] net=${ni} src=(${b % cols},${Math.floor(b / cols)}) expansions=${expansions} treeSize=${tree.size}`)
   266→   266→      // diagnostic : pourquoi le pad de l'arbre est-il scellé ?
   267→   267→      const seedCell = [...tree][0]
   268→   268→      const sb2 = seedCell % cells
   269→   269→      const sx2 = sb2 % cols, sy2 = Math.floor(sb2 / cols)
   270→   270→      const reason = (i: number) => {
   271→   271→        const parts: string[] = []
   272→   272→        if (occupied[i] !== -1 && occupied[i] !== ni) parts.push(`occ=${occupied[i]}`)
   273→   273→        if (dilated[i] !== -1 && dilated[i] !== ni) parts.push(`dil=${dilated[i]}`)
   274→   274→        if (padNet[i] !== -1 && padNet[i] !== ni) parts.push(`pad=${padNet[i]}`)
   275→   275→        if (bodyBlocked[i] && padNet[i] !== ni) parts.push('body')
   276→   276→        if (keepBlock[i] && !isRfNet(ni)) parts.push('keep')
   277→   277→        return parts.join(',') || 'OK'
   278→   278→      }
   279→   279→      console.log(`  seed=(${sx2},${sy2}) layer0 raison=${reason(seedCell)}`)
   280→   280→      for (const [dx, dy] of [[-1, 0], [1, 0], [0, -1], [0, 1]] as const) {
   281→   281→        const nx = sx2 + dx, ny = sy2 + dy
   282→   282→        if (nx < 0 || ny < 0 || nx >= cols || ny >= rows) continue
   283→   283→        console.log(`  L0 voisin (${nx},${ny}): ${reason(idxOf(nx, ny))} | L1: ${reason(cells + idxOf(nx, ny))}`)
   284→   284→      }
   285→   285→    }
   286→   286→    return null
   287→   287→  }

   289→   289→  /* ---- Boucle principale : routage glouton + RIP-UP & REROUTE ---- */
   290→   290→  // Cellules-pad pré-calculées par net (avant le tri : spanOf en dépend)
   291→   291→  const netPinsCellCache = new Map<string, number[]>()
   292→   292→  for (const net of nl.nets) {
   293→   293→    const pinCells: number[] = []
   294→   294→    for (const { ref, pin } of net.pins) {
   295→   295→      const i = padCell.get(`${ref}.${pin}`)
   296→   296→      if (i !== undefined && i >= 0) pinCells.push(i)
   297→   297→    }
   298→   298→    netPinsCellCache.set(net.name, [...new Set(pinCells)])
   299→   299→  }
   300→   300→  const netPinsCells = new Map<number, number[]>()
   301→   301→  for (const net of nl.nets) {
   302→   302→    netPinsCells.set(netIndex.get(net.name)!, netPinsCellCache.get(net.name)!)
   303→   303→  }
   304→   304→
   305→   305→  // Ordre : criticité de classe, puis empan croissant (les bus courts d'abord)
   306→   306→  const spanOf = (n: (typeof nl.nets)[number]) => {
   307→   307→    const cells2 = netPinsCellCache.get(n.name)
   308→   308→    if (!cells2 || cells2.length === 0) return 0
   309→   309→    let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity
   310→   310→    for (const c of cells2) {
   311→   311→      const x = c % cols, y = Math.floor(c / cols)
   312→   312→      if (x < minX) minX = x; if (x > maxX) maxX = x
   313→   313→      if (y < minY) minY = y; if (y > maxY) maxY = y
   314→   314→    }
   315→   315→    return maxX - minX + (maxY - minY)
   316→   316→  }
   317→   317→  const ordered = [...nl.nets]
   318→   318→    .filter((n) => n.pins.length >= 2)
   319→   319→    .sort((a, b) => (CLASS_ORDER[a.cls] ?? 9) - (CLASS_ORDER[b.cls] ?? 9) || spanOf(a) - spanOf(b))

   324→   324→    tree: number[]
   325→   325→    lengthMm: number
   326→   326→  }
   327→   327→  const routedStore = new Map<number, NetRouteState>()
   328→   328→  const failedNets = new Set<number>()
   329→   329→  const triedRip = new Map<number, Set<number>>()
   330→   330→  let done = 0

   332→   332→  /** Reconstruit les masques d'occupation depuis l'état des nets routés */
   333→   333→  function rebuildMasks() {
   334→   334→    occupied.fill(-1)
   335→   335→    dilated.fill(-1)
   336→   336→    routedSet.clear()
   337→   337→    for (const k of routedStore.keys()) routedSet.add(k)
   338→   338→    for (const [ni2, st] of routedStore) markPath(st.tree, ni2, st.vias)

   341→   341→  /** Tente de router entièrement un net (croissance d'arbre, broches les plus proches d'abord) */
   342→   342→  function attemptRoute(ni: number): { ok: boolean; blockedAt?: number } {
   343→   343→    const net = nl.nets[ni]
   344→   344→    const pinCells = netPinsCells.get(ni)!
   345→   345→    const w = widthOfNet(ni)
   346→   346→    const tree = new Set<number>([pinCells[0], pinCells[0] + cells])
   347→   347→    const allSegments: TraceSegment[] = []
   348→   348→    const allVias: Via[] = []
   349→   349→    let netLen = 0
   350→   350→    const remaining = pinCells.slice(1)
   351→   351→
   352→   352→    const nearestFirst = () => {
   353→   353→      let best = 0, bestD = Infinity
   354→   354→      const tcs = [...tree].filter((t) => t < cells)
   355→   355→      for (let i = 0; i < remaining.length; i++) {
   356→   356→        const bx = remaining[i] % cols, by = Math.floor(remaining[i] / cols)
   357→   357→        let d = Infinity
   358→   358→        for (const t of tcs) {
   359→   359→          const ddx = Math.abs((t % cols) - bx), ddy = Math.abs(Math.floor(t / cols) - by)
   360→   360→          if (ddx + ddy < d) d = ddx + ddy
   361→   361→        }
   362→   362→        if (d < bestD) { bestD = d; best = i }
   363→   363→      }
   364→   364→      return best
   365→   365→    }
   366→   366→
   367→   367→    const bx = (v: number) => ({ x: (v % cols) * RES + RES / 2, y: Math.floor((v % cells) / cols) * RES + RES / 2 })
   368→   368→
   369→   369→    while (remaining.length > 0) {
   370→   370→      const pick = nearestFirst()
   371→   371→      const target = remaining.splice(pick, 1)[0]
   372→   372→      // masse & puissance : tentative relâchée d'emblée (elles ont le droit de passer
   373→   373→      // dans les zones de clearance des autres, jamais sur leur cuivre)
   374→   374→      const relaxedFirst = net.cls === 'ground' || net.cls === 'power'
   375→   375→      let path = relaxedFirst ? search(tree, target, ni, true) : search(tree, target, ni)
   376→   376→      if (!path) path = search(tree, target, ni, true) // repli : clearance relâchée
   377→   377→      if (!path) {
   378→   378→        if (process.env.NEXUS_DEBUG) {
   379→   379→          const net2 = nl.nets[ni]
   380→   380→          const pin = net2.pins.find(({ ref, pin: p }) => padCell.get(`${ref}.${p}`) === target)
   381→   381→          const bx2 = target % cols, by2 = Math.floor(target / cols)
   382→   382→          const why = (i: number, relaxed2: boolean) => {
   383→   383→            const parts: string[] = []
   384→   384→            if (occupied[i] !== -1 && occupied[i] !== ni) parts.push(`occ=${occupied[i]}`)
   385→   385→            if (!relaxed2 && dilated[i] !== -1 && dilated[i] !== ni) parts.push(`dil=${dilated[i]}`)
   386→   386→            if (padNet[i] !== -1 && padNet[i] !== ni) parts.push(`pad=${padNet[i]}`)
   387→   387→            if (i >= cells) {
   388→   388→              const p1 = padNetL1[i - cells]
   389→   389→              if (p1 !== -1 && p1 !== ni && !routedSet.has(p1)) parts.push(`res=${p1}`)
   390→   390→            }
   391→   391→            if (keepBlock[i] && !isRfNet(ni) && padNet[i] !== ni) parts.push('keep')
   392→   392→            return parts.join(',') || 'OK'
   393→   393→          }
   394→   394→          console.log(`[route-fail] net=${net2.name} broche=${pin ? `${pin.ref}.${pin.pin}` : '?'} cellule=(${bx2},${by2})`)
   395→   395→          for (const [dx, dy] of [[-1, 0], [1, 0], [0, -1], [0, 1]] as const) {
   396→   396→            const nx2 = bx2 + dx, ny2 = by2 + dy
   397→   397→            if (nx2 < 0 || ny2 < 0 || nx2 >= cols || ny2 >= rows) continue
   398→   398→            console.log(`   L0(${nx2},${ny2}):${why(idxOf(nx2, ny2), true)} | L1:${why(cells + idxOf(nx2, ny2), true)}`)
   399→   399→          }
   400→   400→        }
   401→   401→        return { ok: false, blockedAt: target }
   402→   402→      }
   403→   403→      for (const i of path) tree.add(i)
   404→   404→      // Construction des segments + vias
   405→   405→      let curLayer: 0 | 1 = path[0] < cells ? 0 : 1
   406→   406→      let pts: { x: number; y: number }[] = []
   407→   407→      const flush = (endLayer: 0 | 1) => {
   408→   408→        if (pts.length >= 2) {
   409→   409→          const simplified: { x: number; y: number }[] = [pts[0]]
   410→   410→          for (let i = 1; i < pts.length - 1; i++) {
   411→   411→            const a = simplified[simplified.length - 1], b = pts[i], c = pts[i + 1]
   412→   412→            const collinear = (a.x === b.x && b.x === c.x) || (a.y === b.y && b.y === c.y)
   413→   413→            if (!collinear) simplified.push(b)
   414→   414→          }
   415→   415→          simplified.push(pts[pts.length - 1])
   416→   416→          let len = 0
   417→   417→          for (let i = 1; i < simplified.length; i++)
   418→   418→            len += Math.abs(simplified[i].x - simplified[i - 1].x) + Math.abs(simplified[i].y - simplified[i - 1].y)
   419→   419→          netLen += len
   420→   420→          allSegments.push({ net: net.name, layer: endLayer, pts: simplified, width: w })
   421→   421→        }
   422→   422→      }
   423→   423→      for (let i = 1; i < path.length; i++) {
   424→   424→        const prev = path[i - 1], curr = path[i]
   425→   425→        const layerPrev: 0 | 1 = prev < cells ? 0 : 1
   426→   426→        const layerCur: 0 | 1 = curr < cells ? 0 : 1
   427→   427→        if (layerCur !== layerPrev) {
   428→   428→          pts.push(bx(prev))
   429→   429→          flush(layerPrev)
   430→   430→          const pos = bx(prev)
   431→   431→          allVias.push({ net: net.name, x: Math.round(pos.x * 100) / 100, y: Math.round(pos.y * 100) / 100, drill: rules.minDrill, diameter: rules.minDrill + 0.5 })
   432→   432→          curLayer = layerCur
   433→   433→          pts = [bx(curr)]
   434→   434→        } else {
   435→   435→          if (pts.length === 0) pts.push(bx(prev))
   436→   436→          pts.push(bx(curr))
   437→   437→        }
   438→   438→      }
   439→   439→      flush(curLayer)
   440→   440→    }
   441→   441→
   442→   442→    const seenVia = new Set<string>()
   443→   443→    const uniqVias = allVias.filter((v) => {
   444→   444→      const k = `${v.x},${v.y}`
   445→   445→      if (seenVia.has(k)) return false
   446→   446→      seenVia.add(k); return true
   447→   447→    })
   448→   448→    markPath([...tree], ni, uniqVias)
   449→   449→    routedSet.add(ni)
   450→   450→    routedStore.set(ni, { segments: allSegments, vias: uniqVias, tree: [...tree], lengthMm: netLen })
   451→   451→    return { ok: true }
   452→   452→  }
   453→   453→
   454→   454→  /** Trouve le net routé dont le cuivre bloque le plus la broche en échec
   455→   455→   *  (en excluant les rip-ups déjà tentés et échoués pour ce net) */
   456→   456→  function findBlocker(fi: number, blockedAt: number): number {
   457→   457→    const bx2 = blockedAt % cols, by2 = Math.floor(blockedAt / cols)
   458→   458→    const tried = triedRip.get(fi) ?? new Set<number>()
   459→   459→    let best = -1, bestD = Infinity
   460→   460→    for (const [ri, st] of routedStore) {
   461→   461→      if (ri === fi || tried.has(ri)) continue
   462→   462→      for (let i = 0; i < st.tree.length; i += 2) {
   463→   463→        const c = st.tree[i] % cells
   464→   464→        const d = Math.abs((c % cols) - bx2) + Math.abs(Math.floor(c / cols) - by2)
   465→   465→        if (d < bestD) { bestD = d; best = ri }
   466→   466→      }
   467→   467→    }
   468→   468→    return best
   469→   469→  }
   470→   470→
   471→   471→  /* ---- Passe 1 : routage glouton par criticité ---- */
   472→   472→  for (const net of ordered) {
   473→   473→    if (opts.shouldCancel?.()) break
   474→   474→    const ni = netIndex.get(net.name)!
   475→   475→    const pinCells = netPinsCells.get(ni)!
   476→   476→    if (pinCells.length < 2) {
   477→   477→      done++
   478→   478→      opts.onProgress?.({ done, total: ordered.length, net: net.name, ok: true })
   479→   479→      continue
   480→   480→    }
   481→   481→    const r = attemptRoute(ni)
   482→   482→    if (!r.ok) failedNets.add(ni)
   483→   483→    done++
   484→   484→    opts.onProgress?.({ done, total: ordered.length, net: net.name, ok: r.ok })
   485→   485→  }
   486→   486→
   487→   487→  /* ---- Passe 2 : RIP-UP & REROUTE (jusqu'à 8 rounds, rip multi-bloquants) ---- */
   488→   488→  for (let round = 0; round < 8 && failedNets.size > 0; round++) {
   489→   489→    if (opts.shouldCancel?.()) break
   490→   490→    let progress = false
   491→   491→    for (const fi of [...failedNets]) {
   492→   492→      if (opts.shouldCancel?.()) break
   493→   493→      const probe = attemptRoute(fi)
   494→   494→      const blockedAt = probe.blockedAt
   495→   495→      if (blockedAt === undefined) { failedNets.delete(fi); triedRip.delete(fi); progress = true; continue }
   496→   496→      const blocker = findBlocker(fi, blockedAt)
   497→   497→      if (blocker === -1) continue
   498→   498→      const snapshot = new Map(routedStore)
   499→   499→      const blockerNet = nl.nets[blocker].name
   500→   500→      routedStore.delete(blocker)
   501→   501→      rebuildMasks()
   502→   502→      const rF = attemptRoute(fi)
   503→   503→      if (rF.ok) {
   504→   504→        failedNets.delete(fi)
   505→   505→        triedRip.delete(fi)
   506→   506→        opts.onProgress?.({ done, total: ordered.length, net: `${nl.nets[fi].name} (rip-up de ${blockerNet})`, ok: true })
   507→   507→        const rB = attemptRoute(blocker)
   508→   508→        if (!rB.ok) failedNets.add(blocker)
   509→   509→        progress = true
   510→   510→      } else {
   511→   511→        // Échec : mémorise ce bloquant pour tenter le suivant au prochain round
   512→   512→        if (!triedRip.has(fi)) triedRip.set(fi, new Set())
   513→   513→        triedRip.get(fi)!.add(blocker)
   514→   514→        routedStore.clear()
   515→   515→        for (const [k, v] of snapshot) routedStore.set(k, v)
   516→   516→        rebuildMasks()
   517→   517→      }
   518→   518→    }
   519→   519→    if (!progress) break
   520→   520→  }
   521→   521→
   522→   522→  /* ---- Passe 3 : plan de masse synthétique (copper pour B.Cu) ----
   523→   523→   * Les nets de masse irroutables sont connectés par un plan cuivre généré
   524→   524→   * sur la couche inférieure : toutes les cellules libres, inondées depuis
   525→   525→   * les pads de masse, forment un plan continu (pratique industrielle 2 couches). */
   526→   526→  let groundPour: RoutingSolution['groundPour'] | undefined
   527→   527→  for (const fi of [...failedNets]) {
   528→   528→    if (nl.nets[fi].cls !== 'ground') continue
   529→   529→    const pins = netPinsCells.get(fi)!
   530→   530→    const niG = fi
   531→   531→    // Inondation bicouche : cellules libres L0+L1 reliées par transitions de couche
   532→   532→    const freeCell = (node: number): boolean => {
   533→   533→      const layer = node < cells ? 0 : 1
   534→   534→      const base = node % cells
   535→   535→      if (layer === 0) {
   536→   536→        if (occupied[base] !== -1 || bodyBlocked[base]) return false
   537→   537→        const pn = padNet[base]
   538→   538→        return pn === -1 || pn === niG
   539→   539→      }
   540→   540→      return occupied[cells + base] === -1
   541→   541→    }
   542→   542→    const start0 = pins[0]
   543→   543→    if (!freeCell(start0) || !freeCell(cells + start0)) continue
   544→   544→    const region = new Set<number>([start0, cells + start0])
   545→   545→    const queue: number[] = [start0, cells + start0]
   546→   546→    while (queue.length > 0) {
   547→   547→      const cur = queue.pop()!
   548→   548→      const layer = cur < cells ? 0 : 1
   549→   549→      const base = cur % cells
   550→   550→      const cx = base % cols, cy = Math.floor(base / cols)
   551→   551→      const neighbors: number[] = []
   552→   552→      if (cx > 0) neighbors.push(cur - 1)
   553→   553→      if (cx < cols - 1) neighbors.push(cur + 1)
   554→   554→      if (cy > 0) neighbors.push(cur - cols)
   555→   555→      if (cy < rows - 1) neighbors.push(cur + cols)
   556→   556→      neighbors.push(layer === 0 ? cells + base : base) // via
   557→   557→      for (const nb of neighbors) {
   558→   558→        if (nb < 0 || nb >= total) continue
   559→   559→        if (region.has(nb) || !freeCell(nb)) continue
   560→   560→        region.add(nb)
   561→   561→        queue.push(nb)
   562→   562→      }
   563→   563→    }
   564→   564→    // Tous les pads de masse doivent appartenir au plan
   565→   565→    if (!pins.every((p) => region.has(p))) {
   566→   566→      if (process.env.NEXUS_DEBUG) {
   567→   567→        const reached = pins.filter((p) => region.has(p)).length
   568→   568→        console.log(`[pour] net=${nl.nets[fi].name} région bicouche=${region.size} cellules, pads joints=${reached}/${pins.length} — ÎLOTS SÉPARÉS`)
   569→   569→      }
   570→   570→      continue
   571→   571→    }
   572→   572→    failedNets.delete(fi)
   573→   573→    const toMm = (node: number) => {
   574→   574→      const base = node % cells
   575→   575→      return {
   576→   576→        x: Math.round(((base % cols) * RES + RES / 2) * 100) / 100,
   577→   577→        y: Math.round((Math.floor(base / cols) * RES + RES / 2) * 100) / 100,
   578→   578→      }
   579→   579→    }
   580→   580→    groundPour = {
   581→   581→      top: [...region].filter((n) => n < cells).map(toMm),
   582→   582→      bottom: [...region].filter((n) => n >= cells).map(toMm),
   583→   583→      cols, rows, res: RES,
   584→   584→    }
   585→   585→    opts.onProgress?.({ done, total: ordered.length, net: `${nl.nets[fi].name} (plan de masse synthétique bicouche)`, ok: true })
   586→   586→  }
   587→   587→
   588→   588→  /* ---- Résultats finaux ---- */
   589→   589→  const routes: Route[] = []
   590→   590→  let routedCount = 0
   591→   591→  let totalLen = 0
   592→   592→  let viaCount = 0
   593→   593→  for (const net of ordered) {
   594→   594→    const ni = netIndex.get(net.name)!
   595→   595→    const pinCells = netPinsCells.get(ni)!
   596→   596→    const st = routedStore.get(ni)
   597→   597→    if (st) {
   598→   598→      routedCount++
   599→   599→      totalLen += st.lengthMm
   600→   600→      viaCount += st.vias.length
   601→   601→      routes.push({ net: net.name, segments: st.segments, vias: st.vias, lengthMm: Math.round(st.lengthMm * 10) / 10, routed: true })
   602→   602→    } else if (groundPour && nl.nets[ni].cls === 'ground') {
   603→   603→      routedCount++
   604→   604→      routes.push({ net: net.name, segments: [], vias: [], lengthMm: 0, routed: true, pour: true })
   605→   605→    } else if (pinCells.length < 2) {
   606→   606→      routedCount++
   607→   607→      routes.push({ net: net.name, segments: [], vias: [], lengthMm: 0, routed: true })
   608→   608→    } else {
   609→   609→      routes.push({ net: net.name, segments: [], vias: [], lengthMm: 0, routed: false, failureReason: 'Aucun chemin libre — rip-up épuisé' })
   610→   610→    }
   611→   611→  }
   612→   612→
   613→   613→  return {
   614→   614→    routes,
   615→   615→    routedNets: routedCount,
   616→   616→    totalNets: ordered.length,
   617→   617→    totalLengthMm: Math.round(totalLen * 10) / 10,
   618→   618→    viaCount,
   619→   619→    groundPour,
   620→   620→    durationMs: Date.now() - t0,
   621→   621→  }
   622→   622→}
